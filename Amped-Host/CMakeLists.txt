function(amped_verify_combined_sources)
  get_target_property(_sources amped_recomp SOURCES)
  set(_required "")
  if(AMPED_CAREER_56B40_CANDIDATE)
    list(APPEND _required career_56b40.c)
  endif()
  if(AMPED_REPLAY_SAVE_4F880_CANDIDATE)
    list(APPEND _required replay_save_4f880.c recomp_dispatch_replay_save.c)
  elseif(AMPED_CAREER_56B40_CANDIDATE)
    list(APPEND _required recomp_dispatch_career.c)
  endif()
  foreach(_file IN LISTS _required)
    set(_matches "")
    foreach(_source IN LISTS _sources)
      get_filename_component(_name "${_source}" NAME)
      if(_name STREQUAL _file)
        list(APPEND _matches "${_source}")
      endif()
    endforeach()
    list(LENGTH _matches _count)
    if(NOT _count EQUAL 1)
      message(FATAL_ERROR "Combined build requires exactly one ${_file}; found ${_count}")
    endif()
  endforeach()
endfunction()
